cmake_minimum_required(VERSION 3.16)
project(bs0013_week06_hash_map LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_library(int_hash_map src/int_hash_map.cpp)
target_include_directories(int_hash_map PUBLIC include)
target_compile_options(int_hash_map PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(int_hash_map PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(int_hash_map PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(week06_public_tests tests/public_tests.cpp)
target_link_libraries(week06_public_tests PRIVATE int_hash_map)
target_compile_options(week06_public_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME week06_public_tests COMMAND week06_public_tests)
